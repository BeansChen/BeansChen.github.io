    # 一、安装工作

    1. 安装node，目前版本为 16.20.1

    2. npm init vite
✔ Project name: … react-manage
✔ Select a framework: › React
✔ Select a variant: › TypeScript

    3. 后续还会用到路由和状态管理，所以继续安装

    4.  npm i redux react-redux react-router-dom -S 
 # redux 不一定只支持 react，但是 react-redux 是辅助我们在react中更好使用redux

    5. # 启动项目
npm run dev 

    6. "dev": "vite --host --port 3000", # 用于修改端口

    # 二、项目目录初始化

    打开 src ，除了App.tsx，main.tsx的文件都删除，并删除相关依赖

    # 三、样式初始化

    安装初始化css，比normallize更加简洁

    npm i reset-css -D

    在 main.tsx 中引入，注意引入在 App 之前

    import React from 'react'
import ReactDOM from 'react-dom/client'
// 1.初始化样式
import "reset-css"
// 2.UI框架样式
// ......
// 3.全局样式
// ......
// 4.组件样式
import App from './App.tsx'

    # 四、安装scss

    npm i sass -D

    创建 src/assets/styles/global.scss 文件，内容为

    $color: pink;
body {
  background-color: $color;
  user-select: none; // 禁止选中
}
img{
    // 禁止拖动图片
    -webkit-user-drag: none;
}

    然后引入到 main.tsx 的对应位置中

    import './assets/styles/global.scss'

    # 五、路径别名配置

    import path from "path";

// https://vitejs.dev/config/
export default defineConfig({
  plugins: [react()],
  resolve: {
    alias: {
      "@": path.resolve(__dirname, "./src"),
    },
  },
});

    引入path报错的话，须安装

    npm i -D @types/node

    这个时候虽然可以使用了，但是没有路径提示，在tsconfig.json中：添加两项配置

    "compilerOptions": {
  "baseUrl": "./",
  "paths": {
    "@/*":["src/*"]
  },
}

    # 八、引入路由(旧版本写法)

    1.准备两个页面 ，在 src/views 下创建 Home.tsx 和 About.tsx 两个组件，内容如下：

    const view = () => {
  return <div className="home">home</div>;
};
export default view;

    2.配置对应关系 /src下新建 router文件，再新建index.tsx

    import App from "@/App";
import Home from "@/views/Home";
import About from "@/views/About";
import { BrowserRouter, Routes, Route, Navigate } from "react-router-dom";

// 路由有两种模式 ：BrowserRouter(history模式)，HashRouter（hash模式）

const baseRouter = () => (
  <BrowserRouter>
    <Routes>
      <Route path="/" element={<App />}>
        {/* 重定向 */}
        <Route path="/" element={<Navigate to="home" />}></Route>
        <Route path="/home" element={<Home />}></Route>
        <Route path="/about" element={<About />}></Route>
      </Route>
    </Routes>
  </BrowserRouter>
);
export default baseRouter;

    3.修改main.tsx，把 App 改为Router

    import React from "react";
import ReactDOM from "react-dom/client";
import "reset-css";
import "@/assets/styles/global.scss";
// import App from './App.tsx'
import Router from "./router";

ReactDOM.createRoot(document.getElementById("root")!).render(
  <React.StrictMode>
    <Router />
    {/* <App /> */}
  </React.StrictMode>
);

    4.修改App.tsx

    // import { useState } from 'react'
import { Outlet,Link } from "react-router-dom";
function App() {
  // const [count, setCount] = useState(0)

  return (
    <>
      <div>
        顶级组件
        <Link to="/home">Home</Link>
        <Link to="/about">About</Link>
        {/* 占位符组件，类似于vue中的router-view */}
        <Outlet></Outlet>
      </div>
    </>
  );
}

export default App;

    # 九、路由的第二种写法，路由表

    配置对应关系

    import Home from "@/views/Home";
import About from "@/views/About";

import { Navigate } from "react-router-dom";

const routes = [
  {
    path: "/",
    element: <Navigate to="/home" />,
  },
  {
    path: "/home",
    element: <Home />,
  },
  {
    path: "/about",
    element: <About />,
  },
];

export default routes;

    修改 main.tsx

    import React from "react";
import ReactDOM from "react-dom/client";
import "reset-css";
import "@/assets/styles/global.scss";
import App from "./App.tsx";
import { BrowserRouter } from "react-router-dom";

ReactDOM.createRoot(document.getElementById("root")!).render(
  <React.StrictMode>
    <BrowserRouter>
      <App />
    </BrowserRouter>
  </React.StrictMode>
);

    修改App.tsx

    import { useRoutes, Link } from "react-router-dom";
import router from "./router";
function App() {
  const outlet = useRoutes(router);
  return (
    <>
      <div>
        顶级组件
        <Link to="/home">Home</Link>
        <Link to="/about">About</Link>
        {outlet}
      </div>
    </>
  );
}

export default App;
