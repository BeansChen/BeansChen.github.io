在 JavaScript 原型系统中，constructor 属性是一个非常重要的属性，它指向创建该对象的构造函数。

# 基本概念

function Person(name) {
  this.name = name;
}

const person = new Person('Alice');

console.log(person.constructor === Person); // true
console.log(Person.prototype.constructor === Person); // true

# constructor 的作用

# 1. 标识对象是由哪个构造函数创建的

function Animal(type) {
  this.type = type;
}

const dog = new Animal('dog');
console.log(dog.constructor); // ƒ Animal(type) { this.type = type; }
console.log(dog.constructor.name); // "Animal"

# 2. 用于创建新对象（实例化）

const person1 = new Person('Bob');
const person2 = new person1.constructor('Charlie');
console.log(person2.name); // "Charlie"
console.log(person2 instanceof Person); // true

# 3. 原型继承中的关键环节

function Animal(name) {
  this.name = name;
}

Animal.prototype.eat = function() {
  console.log(this.name + ' is eating');
};

function Dog(name, breed) {
  Animal.call(this, name); // 调用父类构造函数
  this.breed = breed;
}

// 继承原型
Dog.prototype = Object.create(Animal.prototype);

// ⭐ 重要：修复 constructor 指向
Dog.prototype.constructor = Dog;

Dog.prototype.bark = function() {
  console.log('Woof!');
};

const myDog = new Dog('Buddy', 'Golden Retriever');
console.log(myDog.constructor); // ƒ Dog(name, breed) {...}
console.log(myDog instanceof Dog); // true
console.log(myDog instanceof Animal); // true

# constructor 的默认设置

当使用构造函数创建对象时，JavaScript 会自动设置 constructor：

function Car(brand) {
  this.brand = brand;
}

// 默认情况下，原型对象有 constructor 属性指向构造函数
console.log(Car.prototype.constructor === Car); // true

const myCar = new Car('Toyota');
// 实例通过原型链访问 constructor
console.log(myCar.constructor === Car); // true
console.log(myCar.hasOwnProperty('constructor')); // false
console.log(myCar.__proto__.hasOwnProperty('constructor')); // true

# 常见问题和解决方案

# 问题1：手动修改原型时丢失 constructor

function Cat(name) {
  this.name = name;
}

// 直接替换整个原型对象
Cat.prototype = {
  meow: function() {
    console.log('Meow!');
  }
  // 丢失了 constructor 属性
};

const cat = new Cat('Whiskers');
console.log(cat.constructor === Cat); // false
console.log(cat.constructor); // ƒ Object() { [native code] }
// 现在指向 Object 而不是 Cat！

解决方案：

Cat.prototype = {
  constructor: Cat, // 手动添加 constructor
  meow: function() {
    console.log('Meow!');
  }
};

# 问题2：继承时保持正确的 constructor

// 正确做法
function Parent() {}
function Child() {}

Child.prototype = Object.create(Parent.prototype);

// 修复 constructor
Object.defineProperty(Child.prototype, 'constructor', {
  value: Child,
  enumerable: false, // 保持不可枚举
  writable: true,
  configurable: true
});

# 实际应用场景

# 1. 类型检查（替代 instanceof）

function getConstructorName(obj) {
  return obj.constructor.name;
}

const arr = [1, 2, 3];
console.log(getConstructorName(arr)); // "Array"
console.log(arr.constructor === Array); // true

# 2. 克隆对象

function clone(obj) {
  if (obj === null || typeof obj !== 'object') return obj;
  return new obj.constructor(obj);
}

const original = { a: 1, b: 2 };
const copy = clone(original);

# 3. 工厂函数中使用

function createShape(type) {
  const constructors = {
    circle: Circle,
    square: Square,
    triangle: Triangle
  };
  
  const Constructor = constructors[type];
  if (Constructor) {
    return new Constructor();
  }
  return null;
}

# ES6 Class 中的 constructor

ES6 的 class 语法糖也使用了相同的机制：

class Person {
  constructor(name) {
    this.name = name;
  }
  
  greet() {
    console.log(`Hello, I'm ${this.name}`);
  }
}

const person = new Person('Alice');
console.log(person.constructor === Person); // true
console.log(Person.prototype.constructor === Person); // true

# 注意事项

  1. constructor 是可写的

    function A() {}
function B() {}

const a = new A();
a.constructor = B; // 可以修改，但通常不应该这样做
console.log(a.constructor === B); // true
console.log(a instanceof A); // true (instanceof 不受影响)

  2. 不是所有对象都有有意义的 constructor

    const obj = Object.create(null); // 没有原型
console.log(obj.constructor); // undefined

const literalObj = {};
console.log(literalObj.constructor === Object); // true

# 总结

constructor 属性的主要作用：

  • 标识对象的构造函数来源
  • 在原型继承中维护正确的类型链
  • 提供一种创建新对象的方式
  • 辅助进行类型检查和对象创建

在原型继承时，一定要记得修复 constructor 指向，否则会导致类型识别错误。虽然现代开发中很多情况下可以直接使用 instanceof 或 Object.prototype.toString.call()，但理解 constructor 的工作原理对于深入理解 JavaScript 原型系统至关重要。
