# CSS 变量实用技巧与示例

我将为你创建一个展示CSS变量实用技巧的交互式页面，让你直观体验CSS变量的强大功能。

  1. 展示CSS变量的基本定义和使用
  2. 实现主题切换功能
  3. 创建动态调整的UI组件
  4. 演示响应式设计中的应用
  5. 展示CSS变量与JavaScript的交互

        // 主题切换功能
        const themeButtons = document.querySelectorAll('.theme-btn');
        const body = document.body;
        
        themeButtons.forEach(button => {
            button.addEventListener('click', () => {
                // 移除所有主题类
                body.classList.remove('dark-theme', 'warm-theme');
                
                // 更新按钮状态
                themeButtons.forEach(btn => btn.classList.remove('active'));
                button.classList.add('active');
                
                // 应用选中的主题
                const theme = button.getAttribute('data-theme');
                if (theme !== 'default') {
                    body.classList.add(`${theme}-theme`);
                }
            });
        });
        
        // 动态控制CSS变量
        const paddingControl = document.getElementById('padding-control');
        const radiusControl = document.getElementById('radius-control');
        const fontControl = document.getElementById('font-control');
        const opacityControl = document.getElementById('opacity-control');
        const columnsControl = document.getElementById('columns-control');
        
        const paddingValue = document.getElementById('padding-value');
        const radiusValue = document.getElementById('radius-value');
        const fontValue = document.getElementById('font-value');
        const opacityValue = document.getElementById('opacity-value');
        const columnsValue = document.getElementById('columns-value');
        
        const demoElement = document.getElementById('demo-element');
        const responsiveDemo = document.getElementById('responsive-demo');
        
        // 更新CSS变量并显示当前值
        function updateVariable(variableName, value, unit = 'px', displayElement) {
            document.documentElement.style.setProperty(variableName, value + unit);
            if (displayElement) {
                displayElement.textContent = value + (unit === '' ? '' : unit);
            }
        }
        
        // 绑定控件事件
        paddingControl.addEventListener('input', () => {
            updateVariable('--element-padding', paddingControl.value, 'px', paddingValue);
        });
        
        radiusControl.addEventListener('input', () => {
            updateVariable('--element-radius', radiusControl.value, 'px', radiusValue);
        });
        
        fontControl.addEventListener('input', () => {
            updateVariable('--element-font-size', fontControl.value, 'px', fontValue);
        });
        
        opacityControl.addEventListener('input', () => {
            updateVariable('--element-opacity', opacityControl.value, '', opacityValue);
        });
        
        columnsControl.addEventListener('input', () => {
            updateVariable('--responsive-columns', columnsControl.value, '', columnsValue);
        });
        
        // 初始化显示值
        updateVariable('--element-padding', paddingControl.value, 'px', paddingValue);
        updateVariable('--element-radius', radiusControl.value, 'px', radiusValue);
        updateVariable('--element-font-size', fontControl.value, 'px', fontValue);
        updateVariable('--element-opacity', opacityControl.value, '', opacityValue);
        updateVariable('--responsive-columns', columnsControl.value, '', columnsValue);
        
        // 添加一些交互效果
        demoElement.addEventListener('mouseenter', () => {
            demoElement.style.setProperty('--primary-color', 'var(--accent-color)');
        });
        
        demoElement.addEventListener('mouseleave', () => {
            demoElement.style.removeProperty('--primary-color');
        });
        
        // 响应式演示框的交互
        const responsiveBoxes = document.querySelectorAll('.responsive-box');
        responsiveBoxes.forEach((box, index) => {
            box.addEventListener('click', () => {
                // 使用CSS变量设置随机背景色
                const hue = Math.floor(Math.random() * 360);
                box.style.setProperty('--accent-color', `hsl(${hue}, 70%, 60%)`);
                box.textContent = `点击更改颜色 ${index + 1}`;
            });
        });
    </script>

# CSS变量实用技巧总结

这个示例展示了以下CSS变量实用技巧：

  1. 主题切换系统：通过切换CSS类改变整套颜色变量
  2. 动态样式控制：使用JavaScript实时更新CSS变量值
  3. 响应式设计：将CSS变量与媒体查询结合
  4. 一致性设计系统：使用变量维护统一的设计规范
  5. 变量回退值：确保变量未定义时的优雅降级
  6. 作用域控制：在不同选择器中定义和使用变量

# 使用方法

  1. 点击顶部按钮切换不同主题
  2. 使用滑块动态调整示例元素的样式
  3. 调整"列数"滑块查看响应式布局变化
  4. 点击响应式演示框查看动态颜色变化

这个示例完全在单个HTML文件中实现，无需外部依赖，可以直接复制到本地运行。
