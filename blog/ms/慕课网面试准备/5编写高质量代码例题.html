# 一、手写一个JS函数，实现数据扁平化

function flatten(arr: any[]): any[] {
    let res: any[] = [];
    arr.forEach((item) => {
        res = res.concat(item);
    });
    return res;
}

function deepFlatten(arr:any[]):any[]{
    let res:any[] = []
    arr.forEach(item=>{
        if(!Array.isArray(item)){
            res = res.concat(item)
        }else{
            res = res.concat(deepFlatten(item))
        }
    })
    return res
}

# 二、手写一个getType函数，传入任意变量，获取准确类型

function getType(x: any): string {
    const originType = Object.prototype.toString.call(x);
    const spaceIndex = originType.indexOf(" ");
    const type = originType.slice(spaceIndex + 1, -1);
    return type.toLowerCase();
}

# 三、new 一个对象发生了什么，请手写代码展示

Object.create 原理

  1. 判断是否为Object或者null
  2. 创建一个新对象
  3. 将新对象的原型指向传入的对象
  4. 判断第二个参数 Properties 是否存在,存在则为新对象设置属性
  5. 返回新对象
function myCreate(obj, properties) {
  // 判断是否为Object或者null
  if (!(obj instanceof Object) && obj !== null) {
    throw TypeError
  }
  // 创建一个新对象
  let res = {}
  // 将新对象的原型指向传入的对象
  res.__proto__ = obj
  // 判断第二个参数 Properties 是否存在,存在则为新对象设置属性
  if (properties) {
    Object.defineProperties(res, properties)
  }
  // 返回新对象
  return res
}

new 原理

  • 创建一个空对象 obj
  • 使空对象的隐式原型指向构造函数的显式原型
  • 执行构造函数（将obj作为this）
  • 返回obj
function customNew<T>(constructor: Function, ...args: any[]): T {
    // 1.创建一个空对象，继承 constructor 的显式原型
    const obj = Object.create(constructor.prototype);
    // 2.将 obj 作为 this ， 执行 constructor ， 传入参数
    let res = constructor.apply(obj, args);
    // 3.返回该对象实例
    return res instanceof Object ? res : obj;
}

# 四、遍历DOM树

  • 深度优先，递归，贪心
  • 广度优先，使用队列
/**
 * 深度优先遍历
 * @param root
 */
function deepFirstTraverse(root: Node) {
    visitNode(root);
    const childNodes = root.childNodes;
    childNodes.forEach((child) => {
        deepFirstTraverse(child);
    });
}
/**
 * 广度优先遍历：队列实现
 * @param root
 */
function breadFirstTraverse(root: Node) {
    const queue: Node[] = []; // 数组/链表实现队列
    queue.unshift(root);
    while (queue.length > 0) {
        const node = queue.pop();
        if (node == null) break;
        visitNode(node);
        const childNodes = node.childNodes;
        if (childNodes.length) {
            childNodes.forEach((child) => queue.unshift(child));
        }
    }
}
/**
 * 深度优先遍历
 * @param root
 */
function deepFirstTraverse2(root: Node) {
    const stack: Node[] = [];
    stack.push(root); // 根节点压栈
    while (stack.length > 0) {
        const node = stack.pop(); // 出栈
        if (node == null) break;
        visitNode(node);
        // 子节点压栈 reverse反顺序压栈
        const childNodes = node.childNodes;
        if (childNodes.length) {
            Array.from(childNodes)
                .reverse()
                .forEach((child) => stack.push(child));
        }
    }
}
const root = document.getElementById("app");
if (root) {
    deepFirstTraverse(root);
    console.log("----------------------------------------");
    deepFirstTraverse2(root);
    console.log("----------------------------------------");
    breadFirstTraverse(root);
}

# 五、手写一个 lazyMan，实现sleep机制

  • 支持 sleep 和 eat 两个方法
  • 支持链式调用

思路：

注册的方法不是立即调用，而是放入一个任务队列中，依次调用下一个

class lazyMan {
    private name: string;
    private tasks: Function[] = []; // 任务列表
    constructor(name: string) {
        this.name = name;
        setTimeout(() => {
            this.next();
        });
    }
    next() {
        const task = this.tasks.shift();
        if (task) task();
    }
    eat(food: string) {
        const task = () => {
            console.log(`${this.name} 吃 ${food}`);
            this.next();
        };
        this.tasks.push(task);
        return this;
    }
    sleep(delay: number) {
        const task = () => {
            console.log(`${this.name}开始睡觉`);
            setTimeout(() => {
                console.log(
                    `${this.name}已经睡完${delay}分钟，开始执行下一步任务`
                );
                this.next();
            }, delay * 1000);
        };
        this.tasks.push(task);
        return this;
    }
}

const lm = new lazyMan("小明");
lm.eat("香蕉").sleep(2).eat("牛肉").eat("萝卜");

# 六、手写curry函数，实现函数柯里化

  • 提高代码复用性
  • 构建可组合函数
  • 实现延迟执行和懒加载
function curry(fn: Function) {
    const length = fn.length;
    let argsList: any[] = [];
    // ts 中独立的函数 this 需要类型声明
    function calc(this: any, ...args: any[]) {
        argsList = [...argsList, ...args];
        if (argsList.length < length) {
            // 参数不够，返回函数
            return calc;
        } else {
            // 参数够了执行并返回结果
            return fn.apply(this, args.slice(0, length));
        }
    }
    return calc;
}

# 七、instanceof 原理是什么，请用代码展示

**重点：**new 一个对象实际上就是把构造函数的 prototype 赋值给了新对象 __proto__ 属性

function myInstanceOf(instance: any, origin: any): boolean {
    if (instance == null) return false;
    if (typeof instance != "object" && typeof instance != "function") {
        // 值类型都是false
        return false;
    }
    let tempInstance = instance // 返回值修改 instance
    while (tempInstance) {
        if (tempInstance.__proto__ === origin.prototype) {
            return true;
        }else{
            tempInstance = tempInstance.__proto__
        }
    }
    return false;
}

# 九、手写call和apply

重点： 利用了对象调用方法方法的this指向所属对象的原理

Function.prototype.myCall = function (context: any, ...args: []) {
    if (context == null) context = globalThis; // 指向 全局对象
    if (typeof context !== "object") context = new Object(context); // 值类型转换为对象
    const fnKey = Symbol(); // 防止函数名重复
    context[fnKey] = this; // this 就是当前的函数
    const res = context[fnKey](...args); // 绑定了this
    delete context[fnKey]; // 清理fn防止污染
    return res;
};

# 十、手写eventBus

export default class EventBus {
    /**
     * {
     *    'key1': [
     *        { fn: fn1, isOnce: false },
     *        { fn: fn2, isOnce: false },
     *        { fn: fn3, isOnce: true },
     *    ]
     *    'key2': [] // 有序
     *    'key3': []
     * }
     */
    private events: {
        [key: string]: Array<{fn: Function; isOnce: boolean}>
    }

    constructor() {
        this.events = {}
    }

    on(type: string, fn: Function, isOnce: boolean = false) {
        const events = this.events
        if (events[type] == null) {
            events[type] = [] // 初始化 key 的 fn 数组
        }
        events[type].push({ fn, isOnce })
    }

    once(type: string, fn: Function) {
        this.on(type, fn, true)
    }

    off(type: string, fn?: Function) {
        if (!fn) {
            // 解绑所有 type 的函数
            this.events[type] = []
        } else {
            // 解绑单个 fn
            const fnList = this.events[type]
            if (fnList) {
                this.events[type] = fnList.filter(item => item.fn !== fn)
            }
        }
    }

    emit(type: string, ...args: any[]) {
        const fnList = this.events[type]
        if (fnList == null) return

        // 注意
        this.events[type] = fnList.filter(item => {
            const { fn, isOnce } = item
            fn(...args)

            // once 执行一次就要被过滤掉
            if (!isOnce) return true
            return false
        })
    }
}

# 十一、手写LRU缓存

名词解释：

  • LRU= Least Recently Used 最近使用
  • 如果内存优先，只缓存最近使用的
  • 核心API有两个：get set

用Map实现：

  • 哈希表（get set很快）
  • 有序
export default class LRUCache {
    private length: number
    private data: Map<any, any> = new Map()

    constructor(length: number) {
        if (length < 1) throw new Error('invalid length')
        this.length = length
    }

    set(key: any, value: any) {
        const data = this.data

        if (data.has(key)) {
            data.delete(key)
        }
        data.set(key, value) // 保证最近的在最前面

        if (data.size > this.length) {
            // 如果超出了容量，则删除 Map 最老的元素
            const delKey = data.keys().next().value
            data.delete(delKey)
        }
    }

    get(key: any): any {
        const data = this.data

        if (!data.has(key)) return null

        const value = data.get(key)

        data.delete(key)
        data.set(key, value)

        return value
    }
}

数组实现：

实现很简单，就不写了，但有缺陷

  • 移除“沉水”数据，数组shift效率太低
  • get set时移动数据用 splice 效率也太低

双向链表实现：

interface IListNode {
    value: any
    key: string // 存储 key ，方便删除（否则删除时就需要遍历 this.data )
    prev?: IListNode
    next?: IListNode
}

export default class LRUCache {
    private length: number
    private data: { [key: string]: IListNode } = {}
    private dataLength: number = 0
    private listHead: IListNode | null = null
    private listTail: IListNode | null = null

    constructor(length: number) {
        if (length < 1) throw new Error('invalid length')
        this.length = length
    }

    private moveToTail(curNode: IListNode) {
        const tail = this.listTail
        if (tail === curNode) return

        // -------------- 1. 让 prevNode nextNode 断绝与 curNode 的关系 --------------
        const prevNode = curNode.prev
        const nextNode = curNode.next
        if (prevNode) {
            if (nextNode) {
                prevNode.next = nextNode
            } else {
                delete prevNode.next
            }
        }
        if (nextNode) {
            if (prevNode) {
                nextNode.prev = prevNode
            } else {
                delete nextNode.prev
            }

            if (this.listHead === curNode) this.listHead = nextNode
        }

        // -------------- 2. 让 curNode 断绝与 prevNode nextNode 的关系 --------------
        delete curNode.prev
        delete curNode.next

        // -------------- 3. 在 list 末尾重新建立 curNode 的新关系 --------------
        if (tail) {
            tail.next = curNode
            curNode.prev = tail
        }
        this.listTail = curNode
    }

    private tryClean() {
        while (this.dataLength > this.length) {
            const head = this.listHead
            if (head == null) throw new Error('head is null')
            const headNext = head.next
            if (headNext == null) throw new Error('headNext is null')

            // 1. 断绝 head 和 next 的关系
            delete headNext.prev
            delete head.next

            // 2. 重新赋值 listHead
            this.listHead = headNext

            // 3. 清理 data ，重新计数
            delete this.data[head.key]
            this.dataLength = this.dataLength - 1
        }
    }

    get(key: string): any {
        const data = this.data
        const curNode = data[key]

        if (curNode == null) return null

        if (this.listTail === curNode) {
            // 本身在末尾（最新鲜的位置），直接返回 value
            return curNode.value
        }

        // curNode 移动到末尾
        this.moveToTail(curNode)

        return curNode.value
    }

    set(key: string, value: any) {
        const data = this.data
        const curNode = data[key]

        if (curNode == null) {
            // 新增数据
            const newNode: IListNode = { key, value }
            // 移动到末尾
            this.moveToTail(newNode)

            data[key] = newNode
            this.dataLength++

            if (this.dataLength === 1) this.listHead = newNode
        } else {
            // 修改现有数据
            curNode.value = value
            // 移动到末尾
            this.moveToTail(curNode)
        }

        // 尝试清理长度
        this.tryClean()
    }
}

# 十二、实现深拷贝

  • 考虑各种数据类型和循环引用
/**
 * 深拷贝
 * @param obj obj
 * @param map weakmap 为了避免循环引用
 */
export function cloneDeep(obj: any, map = new WeakMap()): any {
    if (typeof obj !== 'object' || obj == null ) return obj

    // 避免循环引用
    const objFromMap = map.get(obj)
    if (objFromMap) return objFromMap

    let target: any = {}
    map.set(obj, target)

    // Map
    if (obj instanceof Map) {
        target = new Map()
        obj.forEach((v, k) => {
            const v1 = cloneDeep(v, map)
            const k1 = cloneDeep(k, map)
            target.set(k1, v1)
        })
    }

    // Set
    if (obj instanceof Set) {
        target = new Set()
        obj.forEach(v => {
            const v1 = cloneDeep(v, map)
            target.add(v1)
        })
    }

    // Array
    if (obj instanceof Array) {
        target = obj.map(item => cloneDeep(item, map))
    }

    // Object
    for (const key in obj) {
        const val = obj[key]
        const val1 = cloneDeep(val, map)
        target[key] = val1
    }

    return target
}
