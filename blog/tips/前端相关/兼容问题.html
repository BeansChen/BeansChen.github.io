# CSS相关兼容性问题

# 1. 浏览器默认样式差异

不同浏览器对标签的默认margin和padding不同。

解决方案：

/* 方案1：使用Normalize.css（推荐） */
/* 访问：https://necolas.github.io/normalize.css/ */

/* 方案2：自定义reset.css */
* {
    margin: 0;
    padding: 0;
    box-sizing: border-box;
}

html, body {
    margin: 0;
    padding: 0;
    -webkit-text-size-adjust: 100%; /* 防止字体缩放 */
}

# 2. 移动端1px边框问题

在高分辨率屏幕上，1px边框会显示过粗。

解决方案：

/* 方案1：使用transform缩放（最常用） */
.thin-border {
    position: relative;
}
.thin-border::after {
    content: '';
    position: absolute;
    top: 0;
    left: 0;
    width: 200%;
    height: 200%;
    border: 1px solid #ccc;
    transform: scale(0.5);
    transform-origin: 0 0;
    pointer-events: none;
    box-sizing: border-box;
}

/* 方案2：使用0.5px（部分浏览器支持） */
@media (-webkit-min-device-pixel-ratio: 2), (min-resolution: 2dppx) {
    .half-pixel-border {
        border: 0.5px solid #ccc;
    }
}

/* 方案3：使用viewport缩放 */
/* 在HTML头部添加： */
<!-- <meta name="viewport" content="width=device-width, initial-scale=0.5, maximum-scale=0.5, minimum-scale=0.5, user-scalable=no"> -->

# 3. placeholder样式不一致

不同浏览器中placeholder的样式（颜色、位置）显示不一致。

解决方案：

input::placeholder {
    color: #999;
    opacity: 1; /* Firefox需要这个 */
    line-height: normal;
    vertical-align: middle;
}

/* 浏览器前缀版本 */
input::-webkit-input-placeholder {
    color: #999;
    line-height: normal;
}

input:-ms-input-placeholder {
    color: #999;
    line-height: normal;
}

input::-moz-placeholder {
    color: #999;
    opacity: 1;
    line-height: normal;
}

/* 特定状态 */
input:focus::placeholder {
    color: #ccc;
}

# 4. 固定定位问题

iOS下fixed元素在软键盘弹出时可能出现定位问题。

解决方案：

/* 方案1：使用sticky定位（现代浏览器） */
.sticky-footer {
    position: sticky;
    bottom: 0;
    z-index: 100;
}

/* 方案2：fixed + 安全区域处理 */
.fixed-element {
    position: fixed;
    bottom: 0;
    bottom: env(safe-area-inset-bottom); /* iOS安全区域 */
    left: 0;
    right: 0;
    padding-bottom: constant(safe-area-inset-bottom); /* 兼容旧版iOS */
    padding-bottom: env(safe-area-inset-bottom);
    background: white;
}

/* 方案3：绝对定位模拟fixed */
@media (max-width: 768px) {
    .mobile-fixed {
        position: absolute;
        width: 100%;
        bottom: 0;
    }
}

# 5. 透明元素点击问题（IE9/IE10）

IE9/IE10中完全透明的元素可能无法点击。

解决方案：

.transparent-element {
    background: rgba(0, 0, 0, 0); /* 推荐 */
    
    /* 兼容IE8-9 */
    background: transparent;
    -ms-filter: "progid:DXImageTransform.Microsoft.gradient(startColorstr=#00000000, endColorstr=#00000000)";
    filter: progid:DXImageTransform.Microsoft.gradient(startColorstr=#00000000, endColorstr=#00000000);
}

# 输入框与键盘处理

# 10. 键盘遮挡输入框

移动端输入框固定在底部时，软键盘可能遮挡输入框。

解决方案：

// 方案1：使用Visual Viewport API（推荐，现代浏览器）
function handleKeyboard() {
    if (!window.visualViewport) return;
    
    const viewport = window.visualViewport;
    let previousHeight = window.innerHeight;
    
    viewport.addEventListener('resize', () => {
        const currentHeight = window.innerHeight;
        
        if (currentHeight < previousHeight) {
            // 键盘显示
            const activeElement = document.activeElement;
            if (activeElement && (activeElement.tagName === 'INPUT' || activeElement.tagName === 'TEXTAREA')) {
                setTimeout(() => {
                    if (activeElement.scrollIntoViewIfNeeded) {
                        activeElement.scrollIntoViewIfNeeded();
                    } else {
                        activeElement.scrollIntoView({ 
                            behavior: 'smooth',
                            block: 'center'
                        });
                    }
                }, 100);
            }
        }
        
        previousHeight = currentHeight;
    });
}

// 方案2：使用resize事件监听
function setupInputFocusHandler() {
    const inputs = document.querySelectorAll('input, textarea');
    const footer = document.getElementById('footer');
    
    inputs.forEach(input => {
        input.addEventListener('focus', () => {
            if (footer) {
                footer.style.position = 'static';
            }
            
            // 滚动到输入框
            setTimeout(() => {
                input.scrollIntoView({ block: 'center' });
            }, 300);
        });
        
        input.addEventListener('blur', () => {
            if (footer) {
                footer.style.position = 'fixed';
            }
            window.scrollTo(0, 0);
        });
    });
}

// 初始化
document.addEventListener('DOMContentLoaded', () => {
    handleKeyboard();
    setupInputFocusHandler();
});

# 19. CSS动画优化

.hardware-accelerated {
    transform: translateZ(0);
    backface-visibility: hidden;
    perspective: 1000px;
}

/* 动画性能优化 */
.animate-element {
    /* 使用transform和opacity */
    transform: translate3d(0, 0, 0);
    opacity: 1;

    /* 告诉浏览器将要变化的属性 */
    will-change: transform, opacity;
}

/* 注意：will-change应该谨慎使用，只在需要时添加 */

/* 禁用动画（性能模式） */
@media (prefers-reduced-motion: reduce) {
    * {
        animation-duration: 0.01ms !important;
        animation-iteration-count: 1 !important;
        transition-duration: 0.01ms !important;
    }
}

# 24. 网络状态检测

解决方案：

// 网络状态检测
function setupNetworkDetection() {
    if (!navigator.onLine) {
        showOfflineMessage();
    }
    
    // 监听网络状态变化
    window.addEventListener('online', () => {
        hideOfflineMessage();
        console.log('网络已连接');
    });
    
    window.addEventListener('offline', () => {
        showOfflineMessage();
        console.log('网络已断开');
    });
    
    // 高级网络信息（需要HTTPS）
    if (navigator.connection) {
        const connection = navigator.connection;
        
        console.log('网络类型:', connection.type);
        console.log('有效类型:', connection.effectiveType);
        console.log('下载速度:', connection.downlink + ' Mbps');
        console.log('往返时间:', connection.rtt + ' ms');
        
        // 监听网络变化
        connection.addEventListener('change', () => {
            console.log('网络状态变化:', {
                type: connection.type,
                effectiveType: connection.effectiveType,
                downlink: connection.downlink
            });
            
            // 根据网络状态调整策略
            if (connection.saveData) {
                enableDataSaverMode();
            }
            
            if (connection.effectiveType === 'slow-2g' || connection.effectiveType === '2g') {
                loadLightVersion();
            }
        });
    }
}

// 离线消息
function showOfflineMessage() {
    const offlineMessage = document.getElementById('offline-message');
    if (offlineMessage) {
        offlineMessage.style.display = 'block';
    }
}

function hideOfflineMessage() {
    const offlineMessage = document.getElementById('offline-message');
    if (offlineMessage) {
        offlineMessage.style.display = 'none';
    }
}

// 数据保护模式
function enableDataSaverMode() {
    // 禁用图片懒加载
    document.querySelectorAll('img[loading="lazy"]').forEach(img => {
        img.loading = 'eager';
    });
    
    // 禁用视频自动播放
    document.querySelectorAll('video[autoplay]').forEach(video => {
        video.autoplay = false;
    });
}

// 轻量版加载
function loadLightVersion() {
    // 替换大图片为小图片
    document.querySelectorAll('img[data-src-small]').forEach(img => {
        const smallSrc = img.getAttribute('data-src-small');
        if (smallSrc) {
            img.src = smallSrc;
        }
    });
    
    // 禁用非必要功能
    const heavyFeatures = document.querySelectorAll('.heavy-feature');
    heavyFeatures.forEach(feature => {
        feature.style.display = 'none';
    });
}

# IE兼容性处理

# 25. IE浏览器支持

注意：微软已于2022年6月15日停止支持IE，新项目应优先考虑现代浏览器。

如果需要支持IE：

<!-- 强制使用最新渲染模式 -->

    <script src="https://cdn.polyfill.io/v2/polyfill.min.js?features=default,fetch,Array.prototype.includes"></script>
<![endif]-->

CSS Hack：

/* IE10+ */
@media all and (-ms-high-contrast: none), (-ms-high-contrast: active) {

    }

    color: red\9; /* IE6-9 */
    *color: blue; /* IE7及以下 */
    _color: green; /* IE6 */
}

# 资源加载优化

# 26. 预加载与预连接

解决方案：

<!-- DNS预解析 -->
<link rel="dns-prefetch" href="//fonts.googleapis.com">
<link rel="dns-prefetch" href="//cdn.example.com">

<!-- 预连接（建立TCP连接） -->
<link rel="preconnect" href="https://fonts.gstatic.com" crossorigin>
<link rel="preconnect" href="https://cdn.example.com">

<!-- 预加载关键资源 -->
<link rel="preload" href="critical.css" as="style">
<link rel="preload" href="critical.js" as="script">
<link rel="preload" href="font.woff2" as="font" type="font/woff2" crossorigin>

<!-- 预获取非关键资源 -->
<link rel="prefetch" href="next-page.html">
<link rel="prefetch" href="next-page-data.js" as="script">

<!-- 延迟加载 -->
<script src="non-critical.js" defer></script>
<script src="analytics.js" async></script>

# 总结

  1. 渐进增强：从基础功能开始，逐步添加高级特性
  2. 优雅降级：确保在老浏览器中基本功能可用
  3. 持续测试：在不同设备和浏览器上测试
  4. 性能优先：始终关注页面加载性能和运行时性能
  5. 用户体验：确保在不同环境下都能提供良好的用户体验
  6. 保持更新：定期更新知识和工具链
