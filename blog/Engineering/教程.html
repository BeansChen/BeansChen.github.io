    Monorepo 教程，基于 Rollup + TypeScript + Turborepo + Jest + VitePress，并配置独立的 pnpm-workspace.yaml。

    # 1. 环境准备

    # 安装 Node.js（>=18.0）和 pnpm
npm install -g pnpm

    # 2. 初始化项目

    # 创建项目目录

    mkdir pangu-monorepo && cd pangu-monorepo
pnpm init

    # 配置独立的 pnpm-workspace.yaml

    packages:
  - 'packages/*'
  - 'shared/*'

    # 修改根目录 package.json

     // 删除了 main 和 test 
{
  "name": "pangu-monorepo",
  "version": "1.0.0",
  "description": "",
  "scripts": {
  },
  "keywords": [],
  "author": "",
  "license": "ISC"
}

    # 3. 创建子包（以 utils 为例）

    # 初始化工具库

    mkdir -p packages/utils/src packages/utils/test
cd packages/utils && pnpm init

    # 安装依赖

    # 根目录执行，安装所有子包共用的基础依赖
pnpm add -Dw @rollup/plugin-commonjs @rollup/plugin-node-resolve @types/jest @types/node jest rollup @rollup/plugin-terser @rollup/plugin-typescript tslib ts-jest typescript vitepress

    # 配置 utils/package.json

    {
  "name": "@cqgm/pangu-utils",
  "version": "1.0.0",
  "description": "工具库",
  "main": "dist/index.cjs.js",
  "module": "dist/index.esm.js",
  "browser": "dist/index.umd.js",
  "types": "dist/types/index.d.ts",
  "files": [
    "dist"
  ],
  "type": "module",
  "scripts": {
    "build": "rollup -c",
    "dev": "rollup -c -w",
    "test": "jest",
    "prepublishOnly": "npm test && npm run build",
    "docs:dev": "vitepress dev docs --port 10002 --open",
    "docs:build": "vitepress build docs"
  },
  "keywords": [],
  "author": "bean",
  "license": "ISC",
  "devDependencies": {}
}

    # 4. 配置 TypeScript 和 Rollup

    # 根目录公共 tsconfig.json

    // tsconfig.base.json
{
  "compilerOptions": {
    "target": "esnext",                                // 编译目标为最新ES标准，保留现代语法
    "module": "esnext",                                // 生成ES模块格式代码
    "declaration": true,                               // 生成 .d.ts 类型声明文件
    "declarationDir": "types",                         // 类型声明文件输出目录
    "moduleResolution": "node",                        // 使用Node.js模块解析策略
    "strict": true,                                    // 启用所有严格类型检查选项
    "sourceMap": true,                                 // 生成sourcemap调试文件
    "resolveJsonModule": true,                         // 允许导入JSON文件
    "esModuleInterop": true,                           // 改善CommonJS/ES模块互操作性
    "lib": ["esnext", "dom"],                          // 包含ES最新特性和DOM类型定义
    "baseUrl": "./",                                   // 设置模块解析基础路径
    "allowJs": true,                                   // 允许编译JavaScript文件
    "forceConsistentCasingInFileNames": true,          // 强制文件名大小写一致(Windows环境建议关闭)
    "allowSyntheticDefaultImports": true,              // 允许无默认导出的模块默认导入
    "noUnusedLocals": true,                            // 禁止未使用的局部变量
    "noUnusedParameters": true,                        // 禁止未使用的函数参数
    "experimentalDecorators": true,                    // 启用装饰器语法支持
    "noImplicitAny": false,                            // 允许隐式any类型(降低严格性)
    "skipLibCheck": true,                              // 跳过声明文件类型检查(提升编译速度)
  }
}

    # 子包继承配置

    // packages/utils/tsconfig.json
{
  "extends": "../../tsconfig.base.json",
  "include": ["src/**/*", "test/**/*", "src/global.d.ts"], // 包含所有 src 目录文件
  "exclude": ["node_modules"] // 排除 node_modules 和测试文件
}

    # Rollup 配置文件

    import typescript from "@rollup/plugin-typescript";
import commonjs from "@rollup/plugin-commonjs";
import resolve from "@rollup/plugin-node-resolve";
import terser from "@rollup/plugin-terser";

export default {
  input: "src/index.ts",
  output: [
    {
      file: 'dist/index.cjs.js',
      format: "cjs",
      exports: "named",
      sourcemap: true,
    },
    {
      file: 'dist/disindex.esm.js',
      format: "es",
      exports: "named",
      sourcemap: true,
    },
    {
      file: 'dist/index.umd.js',
      format: "umd",
      name: "PangUtils",
      exports: "named",
      sourcemap: true,
      globals: {},
    },
  ],
  plugins: [resolve(), commonjs(), typescript({
    tsconfig: "./tsconfig.json",
    compilerOptions: {
      declaration: true,  // 生成声明文件
      outDir: 'dist',  // 输出目录改为dist
      declarationDir: 'dist/types'  // 声明文件输出目录改为dist子目录
    }
  }), terser()],
  external: [],
};

    # 5. 配置 Jest 单元测试

    # 配置 jest.config.js

    export default {
  preset: 'ts-jest',
  testEnvironment: 'node',
  testMatch: ['**/*.test.ts'],
  moduleFileExtensions: ['ts', 'js', 'json'],
  collectCoverage: true,
  collectCoverageFrom: ['src/**/*.ts'],
  coverageDirectory: 'coverage'
};

    # 编写测试示例

    // packages/utils/src/index.ts
/**
 * Sums two numbers.
 * @param a The first number.
 * @param b The second number.
 * @returns The sum of the two numbers.
 */
export const sum = (a: number, b: number) => a + b;

    // packages/utils/test/index.test.ts
import { sum } from '../src/index';

test('adds 1 + 2 to equal 3', () => {
  expect(sum(1, 2)).toBe(3);
});

    # 7. 配置 Turborepo 加速

    # 安装 Turborepo

    # 根目录
pnpm add turbo -Dw

    # 配置 turbo.json

    {
  "$schema": "https://turbo.build/schema.json",
  "tasks": {
    "build": {
      "dependsOn": ["^build"],
      "outputs": ["dist/**"]
    },
    "dev": {
      "cache": false
    },
    "test": {
      "dependsOn": ["build"],
      "outputs": []
    },
    "docs:dev": {
      "cache": false
    }
  }
}

    # 8. 开发与构建命令**

    # 启动所有子包开发

    pnpm dev

    # 构建所有子包

    pnpm build

    # 运行所有测试

    pnpm test

    # 启动文档开发

    pnpm docs:dev

    # 9. 自动化发布（Changesets）

    # 初始化 Changesets

    pnpm add -Dw @changesets/cli
pnpm changeset init

    # 配置选项

    在 .changeset/config.json 中可以配置：

    {
  "$schema": "https://unpkg.com/@changesets/config@3.1.1/schema.json",
  "changelog": "@changesets/cli/changelog",
  "commit": false,
  "fixed": [],
  "linked": [],
  "access": "public",
  "baseBranch": "main",
  "updateInternalDependencies": "patch",
  "ignore": []
}

    # 发布流程

    # 1. 生成变更集
pnpm changeset

# 2. 升级版本号
pnpm changeset version

# 3. 发布到 npm
pnpm changeset publish

    # 10. 最终项目结构

    # 关键优化点

    1. Rollup 打包：每个子包独立配置 Rollup，生成 ESM 和 CJS 格式。
    2. Turborepo 加速：通过缓存和任务管道提升构建、测试效率。
    3. 独立工作区配置：pnpm-workspace.yaml 明确工作区范围。
    4. Jest 测试覆盖：每个子包可单独运行 pnpm test。
    5. 文档与代码绑定：VitePress 文档随代码更新实时同步。

    # 番外篇，添加eslint和prettier

    # 安装

    # 根目录执行
pnpm add -Dw @typescript-eslint/eslint-plugin @typescript-eslint/parser eslint eslint-config-prettier eslint-plugin-prettier prettier globals

    # 配置 eslint.config.js

    import tsESLint from '@typescript-eslint/eslint-plugin';
import tsParser from '@typescript-eslint/parser';
import eslint from '@eslint/js';
import prettierConfig from 'eslint-config-prettier';
import pluginPrettier from 'eslint-plugin-prettier';
import globals from 'globals';
export default [
  eslint.configs.recommended,
  {
    files: ['**/*.ts'],
    languageOptions: {
      parser: tsParser,
      ecmaVersion: 'latest',
      sourceType: 'module',
      globals: {
        ...globals.browser, // 添加浏览器全局变量
        ...globals.node, // 添加Node.js全局变量
        ...globals.jest, // 添加Jest全局变量
        wx: 'readonly', // 解决 wx 未定义问题
        my: 'readonly', // 解决 my 未定义问题
      },
    },
    plugins: {
      '@typescript-eslint': tsESLint,
      prettier: pluginPrettier,
    },
    rules: {
      ...prettierConfig.rules,
      'prettier/prettier': 'error',
      '@typescript-eslint/explicit-function-return-type': 'off',
      'arrow-body-style': 'off',
      'prefer-arrow-callback': 'off',
      '@typescript-eslint/no-unused-vars': [
        'warn',
        {
          argsIgnorePattern: '^_', // 忽略以 _ 开头的未使用参数
          varsIgnorePattern: '^_', // 忽略以 _ 开头的未使用变量
        },
      ],
      '@typescript-eslint/no-explicit-any': 'off',
      '@typescript-eslint/explicit-module-boundary-types': 'off',
      'no-unused-vars': 'off', // 禁用基础规则，使用 TypeScript 的规则
    },
  },
  {
    ignores: ['node_modules', 'dist', 'coverage', '**/*.d.ts'],
  },
];
